# =============================================================================
# Oh My Zsh Configuration
# =============================================================================

export ZSH="/Users/mcclowes/Development/mcclowes/oh-my-zsh"
ZSH_THEME="bureau"

plugins=(
  experiments
  git
  githubutils
  imagetools
  jestutils
  mcclowes
  sublime
  zsh-syntax-highlighting
)

source $ZSH/oh-my-zsh.sh

# =============================================================================
# Environment Variables
# =============================================================================

# Editor
export EDITOR='cursor --wait'

# Python
export PYENV_VERSION="3.7.3"
export PYENV_ROOT="$HOME/.pyenv"

# Node.js
export NVM_DIR="$HOME/.nvm"
export NVM_SYMLINK_CURRENT=true

# =============================================================================
# PATH Configuration
# =============================================================================

# System paths
PATH="/sbin:/bin:/usr/sbin:/usr/bin:/usr/local/sbin:/usr/local/bin:$PATH"

# Homebrew
PATH="/opt/homebrew/bin:/opt/homebrew/sbin:$PATH"

# Python (pyenv)
PATH="$PYENV_ROOT/bin:$PATH"

# gettext
PATH="/usr/local/opt/gettext/bin:$PATH"

# Yarn
PATH="$HOME/.yarn/bin:$HOME/.config/yarn/global/node_modules/.bin:$PATH"

export PATH

# =============================================================================
# Tool Initialization
# =============================================================================

# pyenv
if command -v pyenv 1>/dev/null 2>&1; then
  eval "$(pyenv init -)"
fi

# nvm
[ -s "$(brew --prefix nvm)/nvm.sh" ] && \. "$(brew --prefix nvm)/nvm.sh"
[ -s "$(brew --prefix nvm)/etc/bash_completion.d/nvm" ] && \. "$(brew --prefix nvm)/etc/bash_completion.d/nvm"

# Homebrew completions
if type brew &>/dev/null; then
  FPATH=$(brew --prefix)/share/zsh/site-functions:$FPATH
fi

# =============================================================================
# Custom Functions
# =============================================================================

# Function to check for updates to the current directory if the CD is a git repo
cd () { builtin cd "$@" && cd_git_checker; }
cd_git_checker () { 
  if [ -d .git ]; then
    git fetch
  fi;
}

# npmfetch - Clean and update npm dependencies
function npmfetch {
  rm -rf node_modules && npm update --save-dev && npm update --save
}
